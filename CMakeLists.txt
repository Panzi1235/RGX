cmake_minimum_required(VERSION 3.20)
project(ocrx C CXX)   # CXX solo para enlazar Tesseract (C++); todo el codigo propio es C

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(OCRX_CON_PDFIUM    "Leer PDFs con PDFium"            ON)
option(OCRX_CON_QUIRC     "Leer codigos QR con quirc"       ON)
option(OCRX_CON_IMAGENES  "Abrir JPG/PNG/TIFF (libjpeg-turbo, libpng, libtiff)" ON)
option(OCRX_CON_ZLIB      "Abrir ZIP comprimidos (zlib)"    ON)
option(OCRX_CON_TESSERACT "OCR de texto con Tesseract"       ON)
option(OCRX_RUNTIME_ESTATICO "MSVC: runtime de C dentro del exe (/MT) para no depender del VC++ Redistributable" ON)
option(OCRX_PRUEBAS       "Compilar pruebas"                ON)

if(MSVC)
  add_compile_options(/W4 /utf-8 /wd4996 /wd4100 /wd4127)
  if(OCRX_RUNTIME_ESTATICO)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -Wno-unused-function)
  add_compile_definitions(_POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
endif()

# ---------- dependencias (vcpkg) ----------
find_package(expat CONFIG QUIET)
if(expat_FOUND)
  set(OCRX_LIBS expat::expat)
else()
  find_package(EXPAT REQUIRED)
  set(OCRX_LIBS EXPAT::EXPAT)
endif()
set(OCRX_DEFS "")

if(OCRX_CON_ZLIB)
  find_package(ZLIB REQUIRED)
  list(APPEND OCRX_LIBS ZLIB::ZLIB)
  list(APPEND OCRX_DEFS OCRX_HAVE_ZLIB)
endif()

if(OCRX_CON_QUIRC)
  find_path(QUIRC_INCLUDE quirc.h)
  find_library(QUIRC_LIB NAMES quirc libquirc)
  if(QUIRC_INCLUDE AND QUIRC_LIB)
    list(APPEND OCRX_LIBS ${QUIRC_LIB})
    list(APPEND OCRX_DEFS OCRX_HAVE_QUIRC)
    include_directories(${QUIRC_INCLUDE})
  else()
    message(WARNING "quirc no encontrado: se compila sin lectura de QR")
  endif()
endif()

if(OCRX_CON_IMAGENES)
  find_package(JPEG QUIET)
  if(JPEG_FOUND)
    list(APPEND OCRX_LIBS JPEG::JPEG)
    list(APPEND OCRX_DEFS OCRX_HAVE_JPEG)
  else()
    message(WARNING "libjpeg no encontrada: sin soporte JPG")
  endif()
  find_package(PNG QUIET)
  if(PNG_FOUND)
    list(APPEND OCRX_LIBS PNG::PNG)
    list(APPEND OCRX_DEFS OCRX_HAVE_PNG)
  else()
    message(WARNING "libpng no encontrada: sin soporte PNG")
  endif()
  find_package(TIFF QUIET)
  if(TIFF_FOUND)
    list(APPEND OCRX_LIBS TIFF::TIFF)
    list(APPEND OCRX_DEFS OCRX_HAVE_TIFF)
  else()
    message(WARNING "libtiff no encontrada: sin soporte TIFF")
  endif()
endif()

if(OCRX_CON_TESSERACT)
  find_package(Tesseract CONFIG QUIET)
  if(Tesseract_FOUND)
    list(APPEND OCRX_LIBS Tesseract::libtesseract)
    list(APPEND OCRX_DEFS OCRX_HAVE_TESSERACT)
  else()
    find_path(TESS_INC tesseract/capi.h)
    find_library(TESS_LIB NAMES tesseract tesseract55 tesseract54 tesseract53)
    find_library(LEPT_LIB NAMES leptonica lept)
    if(TESS_INC AND TESS_LIB AND LEPT_LIB)
      include_directories(${TESS_INC})
      list(APPEND OCRX_LIBS ${TESS_LIB} ${LEPT_LIB})
      list(APPEND OCRX_DEFS OCRX_HAVE_TESSERACT)
    else()
      message(WARNING "Tesseract no encontrado: se compila sin OCR de texto (solo QR, XML y PDF con texto)")
    endif()
  endif()
endif()

if(OCRX_CON_PDFIUM)
  # PDFium no esta en vcpkg: se usa el binario de bblanchon/pdfium-binaries descomprimido en third_party/pdfium
  set(PDFIUM_DIR "${CMAKE_SOURCE_DIR}/third_party/pdfium" CACHE PATH "Carpeta con include/ y lib/ (bin/) de pdfium-binaries")
  if(EXISTS "${PDFIUM_DIR}/include/fpdfview.h")
    include_directories("${PDFIUM_DIR}/include")
    if(WIN32)
      list(APPEND OCRX_LIBS "${PDFIUM_DIR}/lib/pdfium.dll.lib")
      set(PDFIUM_DLL "${PDFIUM_DIR}/bin/pdfium.dll")
    else()
      list(APPEND OCRX_LIBS "${PDFIUM_DIR}/lib/libpdfium.so")
    endif()
    list(APPEND OCRX_DEFS OCRX_HAVE_PDFIUM)
  else()
    message(WARNING "PDFium no encontrado en ${PDFIUM_DIR}: se compila sin lectura de PDF. Ejecute scripts/descargar-deps.ps1")
  endif()
endif()

# ---------- biblioteca ----------
file(GLOB_RECURSE OCRX_FUENTES CONFIGURE_DEPENDS
  src/core/*.c src/sunat/*.c src/ingest/*.c src/extraer/*.c src/qr/*.c src/ocr/*.c src/motor.c)
add_library(ocrx_core STATIC ${OCRX_FUENTES})
target_include_directories(ocrx_core PUBLIC include)
target_compile_definitions(ocrx_core PUBLIC ${OCRX_DEFS})
target_link_libraries(ocrx_core PUBLIC ${OCRX_LIBS})
if(NOT WIN32)
  target_link_libraries(ocrx_core PUBLIC m)
endif()

# DLL con API en C (para quien quiera enlazarla directo)
add_library(ocrx SHARED ${OCRX_FUENTES})
target_include_directories(ocrx PUBLIC include)
target_compile_definitions(ocrx PUBLIC ${OCRX_DEFS} PRIVATE OCRX_DLL OCRX_BUILD)
target_link_libraries(ocrx PUBLIC ${OCRX_LIBS})

# ---------- ejecutable: CLI + servicio HTTP + servicio de Windows ----------
add_executable(ocrx_exe cli/ocrx.c src/svc/http.c src/svc/servicio_win.c)
set_target_properties(ocrx_exe PROPERTIES OUTPUT_NAME ocrx)
target_link_libraries(ocrx_exe PRIVATE ocrx_core)
if(WIN32)
  target_link_libraries(ocrx_exe PRIVATE ws2_32 advapi32)
  if(DEFINED PDFIUM_DLL AND EXISTS "${PDFIUM_DLL}")
    add_custom_command(TARGET ocrx_exe POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PDFIUM_DLL}" $<TARGET_FILE_DIR:ocrx_exe>)
  endif()
else()
  target_link_libraries(ocrx_exe PRIVATE pthread)
endif()

# ---------- pruebas ----------
if(OCRX_PRUEBAS)
  enable_testing()
  add_executable(test_sunat tests/test_sunat.c)
  target_link_libraries(test_sunat PRIVATE ocrx_core)
  add_test(NAME reglas_sunat COMMAND test_sunat)
  if(NOT MSVC)
    target_compile_options(test_sunat PRIVATE -Wno-format-zero-length)
  endif()
  add_test(NAME xml_muestra COMMAND $<TARGET_FILE:ocrx_exe> ${CMAKE_SOURCE_DIR}/tests/muestras/factura_F001-123.xml)
  add_test(NAME texto_factura COMMAND $<TARGET_FILE:ocrx_exe> ${CMAKE_SOURCE_DIR}/tests/muestras/factura_texto.txt)
  add_test(NAME texto_ticket COMMAND $<TARGET_FILE:ocrx_exe> ${CMAKE_SOURCE_DIR}/tests/muestras/ticket_ocr.txt)
  add_test(NAME ocr_real_giambar COMMAND $<TARGET_FILE:ocrx_exe> ${CMAKE_SOURCE_DIR}/tests/muestras/ocr_giambar.txt)
  add_test(NAME ocr_real_mundotop COMMAND $<TARGET_FILE:ocrx_exe> ${CMAKE_SOURCE_DIR}/tests/muestras/ocr_mundotop.txt)
endif()

# ---------- instalacion / paquete ----------
install(TARGETS ocrx_exe ocrx RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES ocrx.ini.ejemplo DESTINATION . RENAME ocrx.ini)
install(DIRECTORY models/ DESTINATION models OPTIONAL)
install(DIRECTORY third_party/tessdata/ DESTINATION models/tessdata OPTIONAL)
if(WIN32 AND DEFINED PDFIUM_DLL AND EXISTS "${PDFIUM_DLL}")
  install(FILES "${PDFIUM_DLL}" DESTINATION .)
endif()
